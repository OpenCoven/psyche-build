# Tiered CI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make pull-request CI faster by deduplicating work, parallelizing independent gates, safely skipping unrelated native platforms, and reserving full integration checks for `main` and releases.

**Architecture:** A repository-owned classifier emits fail-open `desktop`, `ios`, and `package` outputs. Independent PR jobs consume those outputs, while aggregate jobs preserve the existing required check names `TypeScript and Rust` and `iOS`. Pushes to `main` enable every platform and add full desktop, iOS app/UI, and package smoke coverage; release verification remains exact-tag and fail-closed.

**Tech Stack:** GitHub Actions YAML, Bash, Git, pnpm, Vitest, TypeScript, Cargo, Xcode/XcodeGen.

> **Reviewed completion amendments:** The detailed original checklist below
> records the implementation sequence. The final reviewed contract supersedes
> its early snippets where they differ: classification uses NUL-delimited
> `--name-status --no-renames` records and includes package sensitivity;
> package smoke runs in its own affected-path job; Windows and Linux run Rust
> test targets rather than check-only builds; iOS-changing pull requests build
> the app/test bundles and run app unit tests; UI execution remains a
> `main`-push escalation; aggregate jobs reject unexpected classifier output;
> and apt receives `DEBIAN_FRONTEND=noninteractive` through `sudo env`.

```ts
import { execFileSync } from 'node:child_process';
import { mkdtempSync, readFileSync, writeFileSync, mkdirSync } from 'node:fs';
import { tmpdir } from 'node:os';
import path from 'node:path';
import { describe, expect, it } from 'vitest';

const classifier = path.resolve('scripts/classify-ci-changes.sh');

function git(cwd: string, ...args: string[]): string {
  return execFileSync('git', args, { cwd, encoding: 'utf8' }).trim();
}

function classify(changes: Record<string, string>, options?: {
  baseSha?: string;
  eventName?: string;
}): Record<string, string> {
  const root = mkdtempSync(path.join(tmpdir(), 'psyche-ci-classifier-'));
  git(root, 'init');
  git(root, 'config', 'user.email', 'ci@example.invalid');
  git(root, 'config', 'user.name', 'CI Test');
  writeFileSync(path.join(root, 'README.md'), 'base\n');
  git(root, 'add', '.');
  git(root, 'commit', '-m', 'base');
  const baseSha = options?.baseSha ?? git(root, 'rev-parse', 'HEAD');

  for (const [relativePath, content] of Object.entries(changes)) {
    const absolutePath = path.join(root, relativePath);
    mkdirSync(path.dirname(absolutePath), { recursive: true });
    writeFileSync(absolutePath, content);
  }
  git(root, 'add', '.');
  git(root, 'commit', '-m', 'change');
  const headSha = git(root, 'rev-parse', 'HEAD');
  const output = path.join(root, 'github-output.txt');

  execFileSync('bash', [classifier], {
    cwd: root,
    env: {
      ...process.env,
      BASE_SHA: baseSha,
      HEAD_SHA: headSha,
      GITHUB_EVENT_NAME: options?.eventName ?? 'pull_request',
      GITHUB_OUTPUT: output,
    },
  });

  return Object.fromEntries(
    readFileSync(output, 'utf8')
      .trim()
      .split('\n')
      .map((line) => line.split('=', 2)),
  );
}

describe('CI change classifier', () => {
  it('skips native jobs for documentation-only changes', () => {
    expect(classify({ 'docs/README.md': 'docs\n' })).toEqual({
      desktop: 'false',
      ios: 'false',
    });
  });

  it('selects only desktop validation for desktop implementation changes', () => {
    expect(classify({ 'native/desktop/example.rs': 'fn main() {}\n' })).toEqual({
      desktop: 'true',
      ios: 'false',
    });
  });

  it('selects both platforms for shared protocol and workflow changes', () => {
    expect(classify({ 'src/daemon/protocol.ts': 'export {};\n' })).toEqual({
      desktop: 'true',
      ios: 'true',
    });
    expect(classify({ '.github/workflows/ci.yml': 'name: CI\n' })).toEqual({
      desktop: 'true',
      ios: 'true',
    });
  });

  it('enables every platform on main pushes', () => {
    expect(classify({ 'docs/README.md': 'docs\n' }, { eventName: 'push' })).toEqual({
      desktop: 'true',
      ios: 'true',
    });
  });

  it('fails open when the base commit cannot be resolved', () => {
    expect(classify(
      { 'docs/README.md': 'docs\n' },
      { baseSha: '0000000000000000000000000000000000000000' },
    )).toEqual({
      desktop: 'true',
      ios: 'true',
    });
  });
});
```

- [ ] **Step 3: Implement the classifier**

Create `scripts/classify-ci-changes.sh`:

```bash
#!/usr/bin/env bash
set -euo pipefail

: "${GITHUB_OUTPUT:?GITHUB_OUTPUT is required}"

desktop=true
ios=true

if [[ "${GITHUB_EVENT_NAME:-}" == "pull_request" ]] \
  && git cat-file -e "${BASE_SHA:-}^{commit}" 2>/dev/null \
  && git cat-file -e "${HEAD_SHA:-}^{commit}" 2>/dev/null; then
  desktop=false
  ios=false

  while IFS= read -r file; do
    case "$file" in
      .github/workflows/*|.github/actions/*|package.json|pnpm-lock.yaml|pnpm-workspace.yaml|rust-toolchain.toml)
        desktop=true
        ios=true
        ;;
      native/desktop/*|frontend/*|packages/*|Cargo.toml|Cargo.lock|__tests__/tauri*|scripts/build-macos-app.mjs)
        desktop=true
        ;;
      native/ios/*|scripts/generate-protocol-fixtures.ts|scripts/app-store-connect.mjs)
        ios=true
        ;;
      src/control/*|src/daemon/*|src/mcp/*|src/orchestration/*|protocol-fixtures/*)
        desktop=true
        ios=true
        ;;
    esac
  done < <(git diff --name-only "${BASE_SHA}...${HEAD_SHA}")
fi

printf 'desktop=%s\n' "$desktop" >> "$GITHUB_OUTPUT"
printf 'ios=%s\n' "$ios" >> "$GITHUB_OUTPUT"
```
